cat > analysis/src/shop_analysis/margin.py <<'EOF'
import numpy as np
from scipy.optimize import curve_fit


def _parabola(k: np.ndarray, a: float, b: float, c: float) -> np.ndarray:
    return a * (k - b) ** 2 + c


def item_scan_margin(k: np.ndarray, sigma2: np.ndarray, flat: float, length: float) -> tuple[float, float]:
    popt, pcov = curve_fit(_parabola, k, sigma2)
    a, _b, c = popt
    eps = np.sqrt(a * c) / (flat**2 * length)
    err = 0.5 * eps * np.sqrt(pcov[0, 0] / a**2 + pcov[2, 2] / c**2)
    return float(eps), float(err)
EOF

cat > analysis/tests/test_margin.py <<'EOF'
import numpy as np
import pytest

from shop_analysis.margin import item_scan_margin


def test_recovers_known_margin():
    k = np.linspace(-2, 2, 21)
    sigma2 = 4.0 * (k - 0.1) ** 2 + 1.0
    eps, err = item_scan_margin(k, sigma2, flat=1.0, length=1.0)
    assert eps == pytest.approx(2.0, rel=1e-6)
    assert err < 1e-6
EOF
